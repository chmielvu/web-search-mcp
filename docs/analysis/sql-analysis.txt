 ```sql
WITH daily AS (
  SELECT date_trunc('day', recorded_at) AS day, provider,
         count() AS calls, sum((status='success')::INT) AS ok
  FROM provider_calls
  WHERE recorded_at >= CURRENT_TIMESTAMP - INTERVAL '14 days'
  GROUP BY day, provider)
SELECT provider, count(DISTINCT day) AS days_active,
       round(avg(ok*100.0/calls)::DOUBLE,1) AS avg_daily_success_pct,
       round(min(ok*100.0/calls)::DOUBLE,1) AS worst_day_pct,
       round(max(ok*100.0/calls)::DOUBLE,1) AS best_day_pct,
       round(stddev(ok*100.0/calls)::DOUBLE,1) AS volatility_stddev,
 ```

 Lens 9 — Domain concentration (HHI)

 ```sql
WITH fr AS (SELECT domain, count() AS hits FROM final_results
            WHERE recorded_at >= CURRENT_TIMESTAMP - INTERVAL '14 days' AND domain IS NOT NULL
            GROUP BY domain),
shares AS (SELECT domain, hits, hits::DOUBLE / sum(hits) OVER () AS share FROM fr)
SELECT (SELECT count() FROM fr) AS distinct_domains,
       (SELECT sum(hits) FROM fr) AS total_hits,
       round(sum(share*share) * 10000, 0) AS HHI,
       CASE WHEN sum(share*share)*10000 < 1500 THEN 'competitive'
            WHEN sum(share*share)*10000 < 2500 THEN 'moderate'
            ELSE 'concentrated' END AS classification
FROM shares;
 ```

 ┌──────────────────┬────────────┬─────┬────────────────┐

 └──────────────────┴────────────┴─────┴────────────────┘

 HHI = 149 (well under 1,500 threshold) → final result domains are very competitive. Top single domain is github.com at 10.81% — high but not dominant. The
